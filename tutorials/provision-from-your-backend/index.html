    Onboard customers from your backend¶

    When a customer signs up for your product you want a ready workspace: an Archy organization and project, an admin who can sign in, a connected model and a few starter agents. This tutorial builds that as one idempotent function that you can call from a signup webhook or a job queue.

    sequenceDiagram
    autonumber
    participant W as Signup webhook
    participant B as Your backend
    participant A as Archy API
    participant M as Customer admin
    W->>B: customer signed up
    B->>B: derive ids from customer id
    B->>A: PUT org, PUT project (service token)
    B->>A: PUT llm-credentials (customer provider key)
    B->>A: POST agents (starter set, pinned model)
    B->>A: POST admins/invite
    A-->>B: invite_link
    B->>M: email the set-password link
    M->>A: reset-password, login

    Prepare: a service token with scopes provision:org provision:project provision:user provision:credential. Locally, mint one as shown in Run Archy locally and add the scopes. Production tokens come from your operator, ideally bound to the organizations you manage (tenant binding).

    Two ideas this tutorial relies on

    1. Ids are derived, not stored. Every Archy id is a UUID computed from your own customer id, so re-running onboarding always hits the same resources.
    2. A service token cannot create LLM credentials or model attachments through the user endpoints (they need an admin user). It pushes credentials through the provisioning endpoint and pins the model inside each agent's config.

    1. The client¶

    # onboarding.py
import os
import uuid
from dataclasses import dataclass

import httpx

NAMESPACE = uuid.UUID("6f1c2f66-8f5b-4c60-9a32-0d8f0a1a2b3c")  # choose once, never change


def archy_id(kind: str, key: str) -> str:
    return str(uuid.uuid5(NAMESPACE, f"{kind}:{key}"))


@dataclass
class Customer:
    id: str                    # your customer id
    name: str
    admin_email: str
    openai_api_key: str | None = None  # present when the customer brings their own key


class Archy:
    def __init__(self) -> None:
        self.http = httpx.Client(
            base_url=os.environ["ARCHY_API_URL"],
            headers={"Authorization": f"Bearer {os.environ['SERVICE_TOKEN']}"},
            timeout=60,
        )

    def call(self, method: str, url: str, **kwargs) -> dict | list:
        response = self.http.request(method, url, **kwargs)
        if response.status_code == 409:
            raise RuntimeError(f"conflict: {response.json()['detail']}")
        response.raise_for_status()
        return response.json()

    2. Starter agents¶

    Describe the agents you want every customer to have. Names are the idempotency key.

    STARTER_AGENTS = [
    {
        "name": "support-assistant",
        "description": "Answers customer questions",
        "prompt": "You are a helpful support assistant for {company}. Be concise and say when you do not know.",
    },
    {
        "name": "summarizer",
        "description": "Summarizes long text",
        "prompt": "Summarize the text you receive in five bullet points for {company}.",
    },
]

    3. Verify the customer's key yourself¶

    Pushed credentials are only used at runtime when they are marked verified. Because there is no user in this flow, you verify the key with a cheap call to the provider before you mark it:

    def key_works(api_key: str) -> bool:
    r = httpx.get("https://api.openai.com/v1/models", headers={"Authorization": f"Bearer {api_key}"}, timeout=20)
    return r.status_code == 200

    If you prefer, push the credential unverified and let the customer's admin call POST /llm-providers/credentials/{id}/verify after signing in.

    4. The onboarding function¶

    def onboard(archy: Archy, customer: Customer, model: str = "gpt-4o-mini") -> dict:
    org_id = archy_id("org", customer.id)
    project_id = archy_id("project", f"{customer.id}/default")
    credential_id = archy_id("credential", f"{customer.id}/openai")
    provisioning = "/api/v1/integrations/provisioning"
    scope = f"/api/v1/orgs/{org_id}/projects/{project_id}"

    # 1. Tenant (idempotent upserts)
    archy.call("PUT", f"{provisioning}/orgs/{org_id}", json={"name": customer.name, "settings": {"customer_id": customer.id}})
    archy.call("PUT", f"{provisioning}/orgs/{org_id}/projects/{project_id}", json={"name": "Default"})

    # 2. Model credential, when the customer brought a key
    model_config: dict = {}
    if customer.openai_api_key:
        archy.call(
            "PUT",
            f"{provisioning}/orgs/{org_id}/llm-credentials",
            json={"credentials": [{
                "credential_id": credential_id,
                "provider": "openai",
                "display_name": "Customer OpenAI",
                "api_key": customer.openai_api_key,
                "is_default": True,
                "is_verified": key_works(customer.openai_api_key),
            }]},
        )
        model_config = {"model": {"provider": "openai", "model_name": model, "credential_id": credential_id}}

    # 3. Starter agents, skipping the ones that exist
    existing = {a["name"] for a in archy.call("GET", f"{scope}/agents")}
    created = []
    for spec in STARTER_AGENTS:
        if spec["name"] in existing:
            continue
        agent = archy.call("POST", f"{scope}/agents", json={
            "name": spec["name"],
            "description": spec["description"],
            "config": {"sys_prompt": spec["prompt"].format(company=customer.name), **model_config},
        })
        created.append(agent["id"])

    # 4. Admin who can sign in (a link only for a new user)
    invite = archy.call("POST", f"{provisioning}/orgs/{org_id}/admins/invite", json={"email": customer.admin_email})

    return {"org_id": org_id, "project_id": project_id, "agents_created": created, "invite_link": invite["invite_link"]}

    Run it twice and the second run creates nothing new: the tenant calls are upserts, the agents are skipped by name, and the invite for an existing admin returns existing_user: true with a null link.

    The pinned model is per agent

    Without an admin user the service token cannot set an organization-wide model attachment. Pinning config.model with a credential_id makes each starter agent use the customer's key. Once the customer's admin signs in they can attach an organization default and remove the pins (see Which model runs).

    5. Wire it to a signup webhook¶

    # main.py
from fastapi import BackgroundTasks, FastAPI
from pydantic import BaseModel

from onboarding import Archy, Customer, onboard

app = FastAPI()
archy = Archy()


class Signup(BaseModel):
    customer_id: str
    company: str
    admin_email: str
    openai_api_key: str | None = None


@app.post("/webhooks/signup", status_code=202)
def signup(event: Signup, background: BackgroundTasks) -> dict:
    def run() -> None:
        result = onboard(archy, Customer(event.customer_id, event.company, event.admin_email, event.openai_api_key))
        send_welcome_email(event.admin_email, result["invite_link"])  # your email code

    background.add_task(run)
    return {"status": "accepted"}

    In production put the call on a queue with retries. Because the function is idempotent, a retry after a partial failure completes the missing steps.

    6. Let the customer's users talk to the agents¶

    Your backend can call the tenant routes with the same service token (see Acting on tenant routes):

    def ask(archy: Archy, customer_id: str, agent_id: str, message: str, session_id: str | None = None) -> dict:
    org_id = archy_id("org", customer_id)
    project_id = archy_id("project", f"{customer_id}/default")
    body = {"message": message, **({"session_id": session_id} if session_id else {})}
    return archy.call("POST", f"/api/v1/agents/{agent_id}/query", params={"org_id": org_id, "project_id": project_id}, json=body)

    Alternatively give users their own tokens: the invite flow, or an identity provider (how users sign in).

    7. Test it without Archy¶

    Mock the transport so tests run offline:

    import httpx

from onboarding import Archy, Customer, onboard


def test_onboard_is_idempotent(monkeypatch):
    calls = []

    def handler(request: httpx.Request) -> httpx.Response:
        calls.append((request.method, request.url.path))
        if request.method == "GET" and request.url.path.endswith("/agents"):
            return httpx.Response(200, json=[{"name": "support-assistant"}, {"name": "summarizer"}])
        if request.url.path.endswith("/admins/invite"):
            return httpx.Response(200, json={"invite_link": None})
        return httpx.Response(200, json={})

    monkeypatch.setenv("ARCHY_API_URL", "http://archy.test")
    monkeypatch.setenv("SERVICE_TOKEN", "t")
    archy = Archy()
    archy.http = httpx.Client(base_url="http://archy.test", transport=httpx.MockTransport(handler))

    result = onboard(archy, Customer("c-1", "Globex", "boss@globex.example"))

    assert result["agents_created"] == []
    assert not [call for call in calls if call[0] == "POST" and call[1].endswith("/agents")]

    Offboarding¶

    Archy has no public endpoint that deletes an organization or a user. Remove customer content with the resource endpoints (DELETE agents, knowledge bases, workflows, credentials) and ask your operator about full tenant removal. DELETE .../llm-credentials/{id} on the provisioning channel is idempotent and safe in an outbox.

    What you learned¶

    You built an idempotent tenant onboarding function with derived ids, a customer-owned model credential, starter agents pinned to it and a first admin invite.
