    Client code¶

    Copy-paste clients for the most common integration tasks. They are small on purpose: read them, keep what you need, and generate a full client from the OpenAPI specification if you want every endpoint.

    Python¶

    Requires httpx (pip install httpx). The class has no other dependency.

    """archy_client.py: a small client for the Archy Agents API."""

from __future__ import annotations

import json
import time
from collections.abc import Iterator
from typing import Any

import httpx


class ArchyError(RuntimeError):
    def __init__(self, response: httpx.Response) -> None:
        try:
            detail = response.json().get("detail")
        except ValueError:
            detail = response.text
        super().__init__(f"{response.request.method} {response.request.url.path} -> {response.status_code}: {detail}")
        self.status_code = response.status_code
        self.detail = detail
        self.request_id = response.headers.get("x-request-id")


class ArchyClient:
    """Acts as one user in one organization and project."""

    def __init__(self, base_url: str, token: str | None = None, *, http: httpx.Client | None = None) -> None:
        self.http = http or httpx.Client(base_url=base_url, timeout=120)
        self.org_id: str | None = None
        self.project_id: str | None = None
        if token:
            self.http.headers["Authorization"] = f"Bearer {token}"

    # --- plumbing --------------------------------------------------------------------------
    def _send(self, method: str, url: str, **kwargs: Any) -> Any:
        response = self.http.request(method, url, **kwargs)
        if response.status_code >= 400:
            raise ArchyError(response)
        return response.json() if response.content else None

    @property
    def scope(self) -> str:
        if not (self.org_id and self.project_id):
            raise RuntimeError("call login() or use_tenant() first")
        return f"/api/v1/orgs/{self.org_id}/projects/{self.project_id}"

    def use_tenant(self, org_id: str, project_id: str) -> None:
        self.org_id, self.project_id = org_id, project_id

    # --- authentication --------------------------------------------------------------------
    def login(self, email: str, password: str, *, remember_me: bool = False) -> dict:
        body = self._send("POST", "/api/v1/auth/login", json={"email": email, "password": password, "remember_me": remember_me})
        self.http.headers["Authorization"] = f"Bearer {body['access_token']}"
        self.use_tenant(body["org_id"], body["project_id"])
        return body

    def switch_tenant(self, org_id: str, project_id: str) -> dict:
        body = self._send("POST", "/api/v1/tenancy/switch", json={"org_id": org_id, "project_id": project_id})
        self.http.headers["Authorization"] = f"Bearer {body['access_token']}"
        self.use_tenant(org_id, project_id)
        return body

    def whoami(self) -> dict:
        return self._send("GET", "/api/v1/tenancy/whoami")

    # --- agents ----------------------------------------------------------------------------
    def create_agent(self, name: str, prompt: str, **config: Any) -> dict:
        return self._send("POST", f"{self.scope}/agents", json={"name": name, "config": {"sys_prompt": prompt, **config}})

    def list_agents(self) -> list[dict]:
        return self._send("GET", f"{self.scope}/agents")

    def ask(self, agent_id: str, message: str, session_id: str | None = None) -> dict:
        body = {"message": message, **({"session_id": session_id} if session_id else {})}
        return self._send("POST", f"/api/v1/agents/{agent_id}/query", json=body)

    def stream_chat(self, agent_id: str, message: str, session_id: str | None = None) -> Iterator[tuple[str, dict]]:
        """Yield (event, data) for start, delta, end and error events."""
        body = {"message": message, **({"session_id": session_id} if session_id else {})}
        with self.http.stream("POST", f"{self.scope}/agents/{agent_id}/chat/stream", json=body, timeout=None) as response:
            if response.status_code >= 400:
                response.read()
                raise ArchyError(response)
            event, data = "message", []
            for line in response.iter_lines():
                if line == "":
                    if data:
                        yield event, json.loads("\n".join(data))
                    event, data = "message", []
                elif line.startswith("event:"):
                    event = line[6:].strip()
                elif line.startswith("data:"):
                    data.append(line[5:].lstrip())

    # --- knowledge bases -------------------------------------------------------------------
    def create_kb(self, name: str, description: str | None = None) -> dict:
        return self._send("POST", f"{self.scope}/knowledge-bases", json={"name": name, "description": description})

    def upload(self, kb_id: str, files: list[tuple[str, bytes]]) -> list[dict]:
        return self._send("POST", f"{self.scope}/knowledge-bases/{kb_id}/documents/upload", files=[("files", f) for f in files])

    def wait_indexed(self, kb_id: str, timeout: float = 600) -> dict:
        deadline = time.monotonic() + timeout
        while time.monotonic() < deadline:
            stats = self._send("GET", f"{self.scope}/knowledge-bases/{kb_id}/stats")
            if stats["files_total"] > 0 and stats["indexing_total"] == 0:
                return stats
            time.sleep(2)
        raise TimeoutError("indexing did not finish")

    def search(self, kb_id: str, query: str, top_k: int = 5, strategy: str = "hybrid") -> list[dict]:
        body = {"query": query, "top_k": top_k, "retrieval_strategy": strategy}
        return self._send("POST", f"{self.scope}/knowledge-bases/{kb_id}/query", json=body)["results"]

    # --- workflows -------------------------------------------------------------------------
    def create_workflow(self, name: str, workflow_type: str, config: dict) -> dict:
        return self._send("POST", f"{self.scope}/workflows", json={"name": name, "type": workflow_type, "config": config})

    def validate(self, workflow_id: str, graph: dict | None = None) -> dict:
        return self._send("POST", f"{self.scope}/workflows/{workflow_id}/validate", json={"graph": graph} if graph else {})

    def run(self, workflow_id: str, message: str, inputs: dict | None = None, timeout: float = 900) -> dict:
        """Start an async run and wait for a final or paused state."""
        started = self._send("POST", f"{self.scope}/workflows/{workflow_id}/runs", json={"message": message, "inputs": inputs})
        delay, deadline = 1.0, time.monotonic() + timeout
        while time.monotonic() < deadline:
            run = self._send("GET", f"{self.scope}/workflows/{workflow_id}/runs/{started['run_id']}")
            if run["status"] in ("succeeded", "failed", "awaiting_input"):
                return run
            time.sleep(delay)
            delay = min(delay * 1.5, 10)
        raise TimeoutError("run still in progress")

    def resume(self, workflow_id: str, session_id: str, answer: str, inputs: dict | None = None) -> dict:
        body = {"answer": answer, **({"inputs": inputs} if inputs else {})}
        return self._send("POST", f"{self.scope}/workflows/{workflow_id}/sessions/{session_id}/resume", json=body)


class Provisioner:
    """Service-token client for creating tenants. Keep it on the server."""

    def __init__(self, base_url: str, service_token: str, *, http: httpx.Client | None = None) -> None:
        self.http = http or httpx.Client(base_url=base_url, timeout=30)
        self.http.headers["Authorization"] = f"Bearer {service_token}"

    def _send(self, method: str, url: str, **kwargs: Any) -> Any:
        response = self.http.request(method, url, **kwargs)
        if response.status_code >= 400:
            raise ArchyError(response)
        return response.json()

    def ensure_org(self, org_id: str, name: str, settings: dict | None = None) -> dict:
        return self._send("PUT", f"/api/v1/integrations/provisioning/orgs/{org_id}", json={"name": name, "settings": settings})

    def ensure_project(self, org_id: str, project_id: str, name: str, description: str | None = None) -> dict:
        return self._send(
            "PUT",
            f"/api/v1/integrations/provisioning/orgs/{org_id}/projects/{project_id}",
            json={"name": name, "description": description},
        )

    def invite_admin(self, org_id: str, email: str, **options: Any) -> dict:
        return self._send("POST", f"/api/v1/integrations/provisioning/orgs/{org_id}/admins/invite", json={"email": email, **options})

    Usage:

    from archy_client import ArchyClient

archy = ArchyClient("http://localhost:8000")
archy.login("admin@acme.example", "a-long-unique-passphrase")

kb = archy.create_kb("handbook")
archy.upload(kb["id"], [("handbook.md", b"Refunds are available within 30 days.")])
archy.wait_indexed(kb["id"])

agent = archy.create_agent("support", "Answer from the knowledge base.", knowledge_base_id=kb["id"])
print(archy.ask(agent["id"], "What is the refund window?")["output"])

for event, data in archy.stream_chat(agent["id"], "And for enterprise customers?"):
    if event == "delta":
        print(data["delta"], end="", flush=True)

    TypeScript¶

    Uses the global fetch of Node 18+ and browsers. Run it on a server: it holds a bearer token.

    // archy.ts
export class ArchyError extends Error {
  constructor(public status: number, public detail: unknown, public requestId?: string) {
    super(`HTTP ${status}: ${typeof detail === "string" ? detail : JSON.stringify(detail)}`);
  }
}

export class Archy {
  private token?: string;
  orgId?: string;
  projectId?: string;

  constructor(private baseUrl: string, token?: string) {
    this.token = token;
  }

  private async send<T>(method: string, path: string, body?: unknown): Promise<T> {
    const response = await fetch(this.baseUrl + path, {
      method,
      headers: {
        ...(this.token ? { Authorization: `Bearer ${this.token}` } : {}),
        ...(body === undefined ? {} : { "Content-Type": "application/json" }),
      },
      body: body === undefined ? undefined : JSON.stringify(body),
    });
    if (!response.ok) {
      const payload = await response.json().catch(() => ({}));
      throw new ArchyError(response.status, payload.detail, response.headers.get("x-request-id") ?? undefined);
    }
    return (await response.json()) as T;
  }

  private get scope(): string {
    if (!this.orgId || !this.projectId) throw new Error("call login() first");
    return `/api/v1/orgs/${this.orgId}/projects/${this.projectId}`;
  }

  async login(email: string, password: string): Promise<void> {
    const body = await this.send<{ access_token: string; org_id: string; project_id: string }>(
      "POST", "/api/v1/auth/login", { email, password });
    this.token = body.access_token;
    this.orgId = body.org_id;
    this.projectId = body.project_id;
  }

  createAgent(name: string, sysPrompt: string, config: Record<string, unknown> = {}) {
    return this.send<{ id: string }>("POST", `${this.scope}/agents`, {
      name, config: { sys_prompt: sysPrompt, ...config } });
  }

  ask(agentId: string, message: string, sessionId?: string) {
    return this.send<{ output: string; session_id: string }>("POST", `/api/v1/agents/${agentId}/query`, {
      message, ...(sessionId ? { session_id: sessionId } : {}) });
  }

  async *streamChat(agentId: string, message: string, sessionId?: string) {
    const response = await fetch(`${this.baseUrl}${this.scope}/agents/${agentId}/chat/stream`, {
      method: "POST",
      headers: { Authorization: `Bearer ${this.token}`, "Content-Type": "application/json" },
      body: JSON.stringify({ message, ...(sessionId ? { session_id: sessionId } : {}) }),
    });
    if (!response.ok) throw new ArchyError(response.status, await response.text());
    const reader = response.body!.pipeThrough(new TextDecoderStream()).getReader();
    let buffer = "";
    for (;;) {
      const { value, done } = await reader.read();
      if (done) return;
      buffer += value;
      let cut: number;
      while ((cut = buffer.indexOf("\n\n")) >= 0) {
        const block = buffer.slice(0, cut);
        buffer = buffer.slice(cut + 2);
        let event = "message";
        const data: string[] = [];
        for (const line of block.split("\n")) {
          if (line.startsWith("event:")) event = line.slice(6).trim();
          else if (line.startsWith("data:")) data.push(line.slice(5).trimStart());
        }
        if (data.length) yield { event, data: JSON.parse(data.join("\n")) };
      }
    }
  }
}

    const archy = new Archy("http://localhost:8000");
await archy.login("admin@acme.example", "a-long-unique-passphrase");
const agent = await archy.createAgent("support", "Be concise.");

  if (event === "delta") process.stdout.write(data.delta);
}

    Shell helpers¶

    # archy.sh: source this file
archy() {  # usage: archy METHOD PATH [json-body]
  local method="$1" path="$2" body="${3:-}"
  if [ -n "$body" ]; then
    curl -sS -X "$method" "$ARCHY_API_URL$path" -H "Authorization: Bearer $ARCHY_TOKEN" \
      -H "Content-Type: application/json" -d "$body"
  else
    curl -sS -X "$method" "$ARCHY_API_URL$path" -H "Authorization: Bearer $ARCHY_TOKEN"
  fi
}
